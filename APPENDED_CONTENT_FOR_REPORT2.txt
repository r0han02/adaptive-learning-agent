6.1 Front-End Module Codes

The front-end of the Carbon Emission Monitoring Web Application is built using Next.js with React and TypeScript, providing real-time session tracking and interactive carbon footprint visualization. The application employs client-side event listeners to capture performance metrics and data transfer volumes during user browsing sessions.

6.1.1 API Route for Session Data Logging

This API endpoint logs real-time session metrics for carbon emission calculation:

```javascript
// app/api/session/track/route.ts
import { db } from '@/lib/db';
import { NextResponse } from 'next/server';

export async function POST(request: Request) {
  try {
    const { userId, dataMb, renderTime, imageCount } = await request.json();
    
    if (!userId) {
      return NextResponse.json({ error: 'User ID required' }, { status: 400 });
    }

    const activity = await db.activities.create({
      data: {
        user_id: userId,
        data_volume_mb: dataMb,
        screen_render_time_ms: renderTime,
        image_count: imageCount,
        timestamp: new Date(),
      },
    });

    return NextResponse.json(activity);
  } catch (error) {
    console.error('Session tracking error:', error);
    return NextResponse.json({ error: 'Database error' }, { status: 500 });
  }
}
```

6.1.2 Carbon Calculation Component

React component for displaying real-time carbon emission estimation:

```typescript
// components/carbonMonitor/CarbonDisplay.tsx
import React, { useState, useEffect } from 'react';
import { calculateCO2 } from '@/lib/carbonCalculator';

interface CarbonMetrics {
  dataMb: number;
  renderTime: number;
  imageCount: number;
  co2Grams: number;
  energyKwh: number;
}

export function CarbonDisplay() {
  const [metrics, setMetrics] = useState<CarbonMetrics>({
    dataMb: 0,
    renderTime: 0,
    imageCount: 0,
    co2Grams: 0,
    energyKwh: 0,
  });

  const [sessionActive, setSessionActive] = useState(true);

  useEffect(() => {
    const interval = setInterval(() => {
      if (sessionActive) {
        // Collect performance metrics
        const perfData = performance.getEntriesByType('navigation')[0];
        const resourceMetrics = performance.getEntriesByType('resource');
        
        let totalDataMb = 0;
        resourceMetrics.forEach(resource => {
          if (resource.transferSize) {
            totalDataMb += resource.transferSize / 1048576; // Convert to MB
          }
        });

        const renderTime = perfData?.domContentLoadedEventEnd - perfData?.domContentLoadedEventStart || 0;
        const imageCount = document.querySelectorAll('img').length;

        const co2Data = calculateCO2(totalDataMb, renderTime, imageCount);

        setMetrics({
          dataMb: totalDataMb,
          renderTime,
          imageCount,
          co2Grams: co2Data.co2,
          energyKwh: co2Data.energy,
        });
      }
    }, 5000); // Update every 5 seconds

    return () => clearInterval(interval);
  }, [sessionActive]);

  return (
    <div className="bg-gradient-to-br from-green-50 to-blue-50 p-6 rounded-lg shadow-lg">
      <h2 className="text-2xl font-bold text-gray-800 mb-4">Your Carbon Footprint</h2>
      
      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
        <div className="bg-white p-4 rounded-lg shadow">
          <p className="text-sm text-gray-600">Data Used</p>
          <p className="text-2xl font-bold text-blue-600">{metrics.dataMb.toFixed(2)} MB</p>
        </div>
        
        <div className="bg-white p-4 rounded-lg shadow">
          <p className="text-sm text-gray-600">Render Time</p>
          <p className="text-2xl font-bold text-purple-600">{metrics.renderTime.toFixed(0)} ms</p>
        </div>
        
        <div className="bg-white p-4 rounded-lg shadow">
          <p className="text-sm text-gray-600">CO₂ Emitted</p>
          <p className="text-2xl font-bold text-red-600">{metrics.co2Grams.toFixed(2)} g</p>
        </div>
        
        <div className="bg-white p-4 rounded-lg shadow">
          <p className="text-sm text-gray-600">Energy Used</p>
          <p className="text-2xl font-bold text-orange-600">{metrics.energyKwh.toFixed(4)} kWh</p>
        </div>
      </div>
    </div>
  );
```

6.1.3 Session Management Hook

Custom hook for managing user sessions and authentication:

```typescript
// hooks/useSession.ts
import { useEffect, useState } from 'react';
import { useRouter } from 'next/navigation';

interface SessionData {
  userId: string;
  email: string;
  loginTime: Date;
  isActive: boolean;
}

export function useSessionManager() {
  const [session, setSession] = useState<SessionData | null>(null);
  const [loading, setLoading] = useState(true);
  const router = useRouter();

  useEffect(() => {
    const fetchSession = async () => {
      try {
        const response = await fetch('/api/auth/session');
        
        if (!response.ok) {
          router.push('/login');
          return;
        }

        const data = await response.json();
        setSession(data);
      } catch (error) {
        console.error('Session fetch failed:', error);
        router.push('/login');
      } finally {
        setLoading(false);
      }
    };

    fetchSession();
  }, [router]);

  const logout = async () => {
    try {
      await fetch('/api/auth/logout', { method: 'POST' });
      router.push('/login');
    } catch (error) {
      console.error('Logout failed:', error);
    }
  };

  return { session, loading, logout };
}
```

6.1.4 CO₂ Popup Alert Component

Component for displaying carbon emission warnings and suggestions:

```typescript
// components/carbonMonitor/CO2Popup.tsx
import React, { useState, useEffect } from 'react';

interface PopupData {
  co2Value: number;
  message: string;
  eco_score: number;
  suggestions: string;
}

export function CO2Popup({ data, onClose }: { data: PopupData; onClose: () => void }) {
  const [visible, setVisible] = useState(true);

  useEffect(() => {
    const timer = setTimeout(() => {
      setVisible(false);
      onClose();
    }, 10000); // Auto-close after 10 seconds

    return () => clearTimeout(timer);
  }, [onClose]);

  if (!visible) return null;

  const severity = data.co2Value > 15 ? 'high' : data.co2Value > 10 ? 'medium' : 'low';
  const bgColor = severity === 'high' ? 'bg-red-100' : severity === 'medium' ? 'bg-yellow-100' : 'bg-green-100';
  const borderColor = severity === 'high' ? 'border-red-500' : severity === 'medium' ? 'border-yellow-500' : 'border-green-500';

  return (
    <div className={`fixed bottom-4 right-4 ${bgColor} border-l-4 ${borderColor} p-4 rounded shadow-lg max-w-sm`}>
      <div className="flex justify-between items-start">
        <div>
          <h3 className="font-bold text-lg mb-2">Session Carbon Report</h3>
          <p className="text-sm mb-2">CO₂ Emitted: <span className="font-bold">{data.co2Value.toFixed(2)} g</span></p>
          <p className="text-sm mb-2">Eco Score: <span className="font-bold">{data.eco_score}/100</span></p>
          <p className="text-sm text-gray-700 mb-2">{data.message}</p>
          <p className="text-xs text-gray-600 italic">{data.suggestions}</p>
        </div>
        <button onClick={() => setVisible(false)} className="text-gray-500 hover:text-gray-700">✕</button>
      </div>
    </div>
  );
}
```

6.2.3 Carbon Calculation Algorithm

Core logic for calculating CO₂ emissions from session metrics:

```typescript
// lib/carbonCalculator.ts
export interface CarbonMetrics {
  co2: number;
  energy: number;
  ecoScore: number;
  suggestion: string;
}

export function calculateCO2(dataMb: number, renderTime: number, imageCount: number): CarbonMetrics {
  // Constants (based on industry studies)
  const CO2_PER_MB = 0.81; // grams CO₂ per MB of data transferred
  const CO2_PER_RENDER_MS = 0.0015; // grams CO₂ per millisecond of render time
  const CO2_PER_IMAGE = 0.5; // grams CO₂ per image loaded
  const ENERGY_PER_CO2 = 0.00044; // kWh per gram of CO₂

  // Calculate individual components
  const dataCo2 = dataMb * CO2_PER_MB;
  const renderCo2 = renderTime * CO2_PER_RENDER_MS;
  const imageCo2 = imageCount * CO2_PER_IMAGE;

  // Total CO₂ emissions in grams
  const totalCo2 = dataCo2 + renderCo2 + imageCo2;

  // Energy consumption in kWh
  const energyKwh = totalCo2 * ENERGY_PER_CO2;

  // Eco score (0-100, where higher is better)
  const baseEcoScore = 100 - (totalCo2 * 2);
  const ecoScore = Math.max(0, Math.min(100, baseEcoScore));

  // Generate suggestions based on metrics
  let suggestion = 'Good session! Keep up the green practices.';
  if (dataMb > 50) {
    suggestion = 'High data usage detected. Consider optimizing image loads or disabling auto-play videos.';
  } else if (renderTime > 300) {
    suggestion = 'Slow rendering detected. Your device may be struggling. Consider closing other tabs or reducing visual effects.';
  } else if (imageCount > 100) {
    suggestion = 'Heavy image loading. Use image optimization techniques or lazy loading to reduce emissions.';
  }

  return {
    co2: totalCo2,
    energy: energyKwh,
    ecoScore,
    suggestion,
  };
}
```

6.2.4 Error Handling and Middleware

Middleware for handling database errors and request logging:

```typescript
// middleware.ts
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

export function middleware(request: NextRequest) {
  try {
    // Add request ID for tracing
    const requestHeaders = new Headers(request.headers);
    requestHeaders.set('x-request-id', crypto.randomUUID());
    requestHeaders.set('x-request-time', new Date().toISOString());

    // Log protected routes
    if (request.nextUrl.pathname.startsWith('/dashboard')) {
      console.log(`Protected route accessed: ${request.nextUrl.pathname}`);
    }

    return NextResponse.next({
      request: {
        headers: requestHeaders,
      },
    });
  } catch (error) {
    console.error('Middleware error:', error);
    return NextResponse.error();
  }
}

export const config = {
  matcher: ['/api/:path*', '/dashboard/:path*'],
};
```

7.4 Challenges and Solutions

**Challenge 1: Accurate CO₂ Calculation**
Initial difficulty in establishing accurate conversion factors between network data transfer and carbon emissions. Solved by incorporating industry-standard metrics based on research from Green Software Foundation.

**Challenge 2: Real-Time Performance Tracking**
Capturing granular performance metrics without impacting page load times. Resolved by using non-blocking JavaScript operations and the Performance Observer API for asynchronous metric collection.

**Challenge 3: Database Scalability**
Ensuring the system handles high-frequency activity logging without database bottlenecks. Addressed through strategic indexing, connection pooling, and partitioning of historical data.

**Challenge 4: User Privacy**
Balancing detailed session tracking with user privacy concerns. Implemented through data anonymization options, transparent privacy policies, and secure session management.
